cmake_minimum_required(VERSION 3.20)
project(lob-engine VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(LOB_BUILD_TESTS "Build unit tests" ON)
option(LOB_BUILD_BENCH "Build benchmarks" ON)
option(LOB_BUILD_TOOLS "Build tools" ON)
option(LOB_SANITIZE "Enable ASan + UBSan" OFF)
option(LOB_WERROR "Treat warnings as errors" OFF)

add_library(lob_options INTERFACE)
if(MSVC)
  target_compile_options(lob_options INTERFACE /W4)
else()
  target_compile_options(lob_options INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
  if(LOB_WERROR)
    target_compile_options(lob_options INTERFACE -Werror)
  endif()
  if(LOB_SANITIZE)
    target_compile_options(lob_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
    target_link_options(lob_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(lob src/version.cpp)
target_include_directories(lob PUBLIC include)
target_link_libraries(lob PRIVATE lob_options)

if(LOB_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(LOB_BUILD_BENCH)
  add_subdirectory(bench)
endif()
if(LOB_BUILD_TOOLS)
  add_subdirectory(tools)
endif()
