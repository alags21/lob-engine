find_package(ZLIB QUIET)

add_library(lob_itchlib STATIC itch_validate.cpp itch_synth.cpp)
target_include_directories(lob_itchlib PUBLIC .)
target_link_libraries(lob_itchlib PUBLIC lob lob_reference PRIVATE lob_options)

add_executable(lob_itch main.cpp bench.cpp)
target_include_directories(lob_itch PRIVATE ${PROJECT_SOURCE_DIR}/bench)  # platform.hpp (clock, pinning)
target_link_libraries(lob_itch PRIVATE lob_itchlib lob_options)
if(ZLIB_FOUND)
  target_compile_definitions(lob_itch PRIVATE LOB_HAVE_ZLIB=1)
  target_link_libraries(lob_itch PRIVATE ZLIB::ZLIB)
else()
  message(STATUS "zlib not found: lob_itch cannot read .gz files directly")
endif()

if(LOB_BUILD_TESTS)
  find_package(GTest REQUIRED)
  add_executable(lob_itch_tests test_itch.cpp)
  target_link_libraries(lob_itch_tests PRIVATE lob_itchlib lob_options GTest::gtest_main)
  gtest_discover_tests(lob_itch_tests)
endif()
